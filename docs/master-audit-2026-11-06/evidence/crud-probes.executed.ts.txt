const results:any[]=[];let owner:any,cashier:any,warehouse:any;
async function request(url:string,actor:any,body:any,method='POST') {return new NextRequest('http://localhost'+url,{method,headers:{'content-type':'application/json',...(actor?{authorization:'Bearer '+await createSessionToken(actor)}:{}),'idempotency-key':'audit-duplicate-request'},body:JSON.stringify(body)});}
async function probe(id:string,title:string,run:()=>Promise<any>){await executeSeed(prisma);owner=await prisma.user.findFirst({where:{role:Role.OWNER}});cashier=await prisma.user.findFirst({where:{role:Role.CASHIER}});warehouse=await prisma.user.findFirst({where:{role:Role.WAREHOUSE}});try{const result={id,title,...await run()};results.push(result);console.log(JSON.stringify(result));}catch(e){const result={id,title,status:'PROBE_ERROR',error:String(e)};results.push(result);console.log(JSON.stringify(result));}}

await probe('B01','Cashier cannot create an Owner account',async()=>{const res=await userPOST(await request('/api/users',cashier,{name:'Audit Owner',email:'audit-owner@example.test',role:'OWNER',password:'AuditTestOnly1!'}));const data=await res.json();return {expected:'403, no new account',actual:{http:res.status,createdRole:data.user?.role},status:res.status===403?'PASS':'FAIL'};});
await probe('B02','Anonymous company edits and deletion are rejected',async()=>{const file=path.join(process.cwd(),'prisma/audit-company.db');writeFileSync(file,'disposable audit fixture');saveCompanies([DEFAULT_COMPANY,{...DEFAULT_COMPANY,id:'audit-company',dbFile:'audit-company.db',isDefault:false}]);const edit=await companyPATCH(await request('/api/companies',null,{id:'audit-company',name:'Anonymous edit'},'PATCH'));const del=await companyDELETE(new NextRequest('http://localhost/api/companies?id=audit-company',{method:'DELETE'}));return {expected:'401 for both requests, fixture file retained',actual:{editHttp:edit.status,deleteHttp:del.status,fileExists:existsSync(file)},status:edit.status===401&&del.status===401?'PASS':'FAIL'};});
await probe('B03','Anonymous demo seed cannot create posted documents',async()=>{const before={orders:await prisma.posOrder.count(),journals:await prisma.journalEntry.count()};const res=await demoPOST();const after={orders:await prisma.posOrder.count(),journals:await prisma.journalEntry.count()};return {expected:'401; database unchanged',actual:{http:res.status,before,after},status:res.status===401?'PASS':'FAIL'};});
await probe('B04','Stock adjustments create movement and valuation journal',async()=>{const before={moves:await prisma.stockMove.count(),inventory:(await prisma.account.findUnique({where:{code:'1030'}}))?.currentBalance};const res=await stockPATCH(await request('/api/products/RICE-CHICK-S',warehouse,{locationCode:'WH-JKT',newOnHand:41},'PATCH'),{params:Promise.resolve({sku:'RICE-CHICK-S'})});const after={moves:await prisma.stockMove.count(),inventory:(await prisma.account.findUnique({where:{code:'1030'}}))?.currentBalance};return {expected:'Movement +1, Inventory +26000, balanced adjustment journal',actual:{http:res.status,before,after},status:after.moves>before.moves&&after.inventory===before.inventory!+26000?'PASS':'FAIL'};});
await probe('B05','Cashier product creation cannot bypass operations permission',async()=>{const res=await productPOST(await request('/api/products',cashier,{sku:'AUDIT-NEGATIVE',name:'Audit Product',salesPrice:-100,standardCost:-50,defaultVendorId:'C01',initialWhStock:7}));const product=await prisma.productVariant.findUnique({where:{sku:'AUDIT-NEGATIVE'},include:{stockLevels:true}});return {expected:'403; no product',actual:{http:res.status,price:product?.salesPrice,cost:product?.standardCost,vendor:product?.defaultVendorId,stock:product?.stockLevels.map(s=>({location:s.locationCode,onHand:s.onHand}))},status:res.status===403?'PASS':'FAIL'};});
await probe('B06','Customer contact fields persist and opening balance cannot be minted',async()=>{const before=(await prisma.account.findUnique({where:{code:'1020'}}))?.currentBalance;const res=await contactPOST(await request('/api/contacts',owner,{name:'Audit Contact',type:'CUSTOMER',customerType:'INDIVIDUAL',city:'Jakarta',email:'audit@example.test',phone:'080000000',address:'Audit address',openReceivable:123456}));const data=await res.json();const fresh=new PrismaClient({datasourceUrl:process.env.DATABASE_URL});const saved:any=await fresh.contact.findUnique({where:{id:data.contact?.id}});await fresh.$disconnect();return {expected:'Contact details persist; opening receivable requires balanced migration workflow',actual:{http:res.status,reopenedName:saved?.name,email:saved?.email??null,phone:saved?.phone??null,address:saved?.address??null,openingAR:saved?.openReceivable,ledgerBefore:before,ledgerAfter:(await prisma.account.findUnique({where:{code:'1020'}}))?.currentBalance},status:'FAIL',persistenceCheck:saved?.name==='Audit Contact'?'PASS':'FAIL'};});
await probe('B07','CRM creation validates Proposition minimum fields',async()=>{const res=await leadPOST(await request('/api/crm/leads',cashier,{name:'Audit invalid opportunity',stage:'PROPOSITION',expectedRevenue:0}));const data=await res.json();return {expected:'403 for Cashier or 400 invalid lead',actual:{http:res.status,stage:data.lead?.stage,customer:data.lead?.contactId,revenue:data.lead?.expectedRevenue},status:res.status>=400?'PASS':'FAIL'};});
await probe('B08','Negative sales quantities are rejected before reservation',async()=>{let blocked=false;try{const order=await createSalesQuotation({contactId:'C01',lines:[{productSku:'RICE-CHICK-S',quantity:-2,unitPrice:48000}],actor:owner});await confirmSalesOrder(order.id,owner);}catch{blocked=true;}const stock=await prisma.stockLevel.findUnique({where:{productSku_locationCode:{productSku:'RICE-CHICK-S',locationCode:'WH-JKT'}}});return {expected:'Reject quantity -2; reserved remains 6',actual:{blocked,reserved:stock?.reserved},status:blocked&&stock?.reserved===6?'PASS':'FAIL'};});
await probe('B09','Draft line edit enforces maximum discount of 20 percent',async()=>{const order=await createSalesQuotation({contactId:'C01',lines:[{productSku:'RICE-CHICK-S',quantity:1,unitPrice:48000}],actor:owner});let blocked=false,result:any;try{result=await updateSalesOrderLine({orderId:order.id,lineId:order.lines[0].id,discountPercent:50,actor:cashier});}catch{blocked=true;}return {expected:'Reject unauthorized 50 percent discount',actual:{blocked,discount:result?.discountPercent},status:blocked?'PASS':'FAIL'};});
await probe('B10','POS respects session location and reserved inventory',async()=>{const session=await openPosSession({cashierId:cashier.id,actor:cashier});const res=await posPOST(await request('/api/pos/orders',cashier,{sessionId:session.id,locationCode:'WH-JKT',lines:[{productSku:'RICE-CHICK-S',quantity:35}]}));const stock=await prisma.stockLevel.findUnique({where:{productSku_locationCode:{productSku:'RICE-CHICK-S',locationCode:'WH-JKT'}}});return {expected:'Reject WH sale with CAFE session; WH stock40/reserved6 unchanged',actual:{http:res.status,onHand:stock?.onHand,reserved:stock?.reserved},status:res.status>=400&&stock?.onHand===40?'PASS':'FAIL'};});
await probe('B11','Cashier cannot validate warehouse transfers',async()=>{const res=await transferPOST(await request('/api/inventory/transfers',cashier,{sourceLocation:'WH-JKT',destLocation:'CAFE-JKT',lines:[{productSku:'COF-LATT-ICE',quantity:1}]}));const data=await res.json();let validateHttp:number|undefined;if(data.transfer){const validate=await validateTransferPOST(await request('/api/inventory/transfers/'+data.transfer.id+'/validate',cashier,{}),{params:Promise.resolve({id:data.transfer.id})});validateHttp=validate.status;}return {expected:'403 on create and validate',actual:{createHttp:res.status,validateHttp},status:res.status===403?'PASS':'FAIL'};});
await probe('B12','POS repeated request is posted exactly once',async()=>{const session=await openPosSession({cashierId:cashier.id,actor:cashier});const body={sessionId:session.id,lines:[{productSku:'COF-LATT-HOT',quantity:1}]};const first=await posPOST(await request('/api/pos/orders',cashier,body));const retry=await posPOST(await request('/api/pos/orders',cashier,body));const count=await prisma.posOrder.count();return {expected:'One order for repeated idempotency key',actual:{first:first.status,retry:retry.status,orders:count},status:count===1?'PASS':'FAIL'};});
